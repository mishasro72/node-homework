SELECT order_id, SUM(price * quantity) AS total_price FROM line_items JOIN products ON line_items.product_id = products.product_id GROUP BY order_id ORDER BY order_id LIMIT 5;

SELECT customer_name, AVG(total_price) AS average_order_price FROM customers AS c JOIN (SELECT o.customer_id, o.order_id, SUM(products.price * li.quantity) AS total_price FROM orders AS o JOIN line_items AS li ON o.order_id=li.order_id JOIN products ON li.product_id = products.product_id GROUP BY o.customer_id, o.order_id) AS t ON c.customer_id = t.customer_id GROUP BY c.customer_id, c.customer_name ORDER BY c.customer_name;
