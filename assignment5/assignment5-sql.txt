SELECT o.order_id, SUM(p.price * li.quantity) AS total_price FROM (SELECT order_id FROM orders ORDER BY order_id LIMIT 5) AS o JOIN line_items AS li ON o.order_id = li.order_id JOIN products AS p ON li.product_id = p.product_id GROUP BY o.order_id ORDER BY o.order_id;

SELECT c.customer_name, AVG(t.total_price) AS average_order_price FROM customers AS c JOIN (SELECT o.order_id, o.customer_id, SUM(p.price * li.quantity) AS total_price FROM (SELECT order_id, customer_id FROM orders) AS o JOIN line_items AS li ON o.order_id=li.order_id JOIN products AS p ON li.product_id = p.product_id GROUP BY o.customer_id, o.order_id) AS t ON c.customer_id = t.customer_id GROUP BY c.customer_id, c.customer_name ORDER BY c.customer_name;
